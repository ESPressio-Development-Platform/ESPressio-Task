cmake_minimum_required(VERSION 3.16)
project(ESPressioTaskContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ESPRESSIO_SYSTEM_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../ESPressio-System" CACHE PATH "System primitives_redesign checkout")
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
foreach(test idle_worker executor configuration)
    add_executable(${test}_tests ${test}_tests.cpp)
    target_include_directories(${test}_tests PRIVATE ../src ${ESPRESSIO_SYSTEM_SOURCE_DIR}/src)
    target_compile_options(${test}_tests PRIVATE -Wall -Wextra -Werror -fno-rtti)
    target_link_libraries(${test}_tests PRIVATE Threads::Threads)
    add_test(NAME ${test} COMMAND ${test}_tests)
endforeach()
add_test(NAME rejected_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_compile_fail.py
    ${CMAKE_CXX_COMPILER} ${CMAKE_CURRENT_SOURCE_DIR}/../src ${ESPRESSIO_SYSTEM_SOURCE_DIR}/src)
